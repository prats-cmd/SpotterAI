function showApiError(msg, status) { let banner = document.getElementById('apiBanner'); if (!banner) { banner = document.createElement('div'); banner.id = 'apiBanner'; banner.style.cssText = 'background:#FF6A4D;color:#0E0F0A;font-size:13px;font-weight:700;padding:9px 16px;text-align:center;border-radius:8px;margin:0 1.25rem 0.5rem;line-height:1.4;'; const bar = document.querySelector('.bottom-action-bar'); bar.parentNode.insertBefore(banner, bar); } banner.style.display = 'block'; banner.textContent = '⚠️ ' + ( status === 401 ? 'Invalid NVIDIA key — set NEMOTRON_API_KEY or NVIDIA_API_KEY in Vercel env vars.' : status === 429 ? 'Rate limited or out of NVIDIA credits.' : msg ); addMessage('⚠️ ' + banner.textContent, 'ai'); }
